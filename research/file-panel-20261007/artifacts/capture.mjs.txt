async function open(w, h, scheme) {
	const page = await browser.newPage({ viewport: { width: w, height: h }, colorScheme: scheme })
	page.on('pageerror', (e) => logs.push('PAGEERROR: ' + e.message))
	await page.goto('http://127.0.0.1:5173/preview')
	await page.waitForTimeout(1200)
	const nav = page.getByRole('button', { name: /Refine navigation/ }).first()
	if (!(await nav.isVisible())) { await page.keyboard.press('Control+b'); await page.waitForTimeout(500) }
	await nav.click(); await page.keyboard.press('Control+b').catch(() => {})
	await page.waitForTimeout(1200)
	return page
}
async function quick(page, name) {
	await page.getByRole('tab', { name: 'Open a file' }).click()
	await page.waitForTimeout(300)
	await page.keyboard.type(name)
	await page.waitForTimeout(800)
	await page.keyboard.press('Enter')
	await page.waitForTimeout(1000)
}
